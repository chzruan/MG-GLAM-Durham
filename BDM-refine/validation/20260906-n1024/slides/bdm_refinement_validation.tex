\begin{frame}{Fix halo identity and the bound population}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\highlight{Duplicate survival}
\begin{itemize}
\item Equal-mass candidates can have slightly different particle sets.
\item Remove exact member-set copies.
\item Break mass ties by stable host priority; exclude inner centres.
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\highlight{Consistent properties}
\begin{itemize}
\item Use the outermost supported SO crossing.
\item Iterate unbinding until membership converges.
\item Measure properties from the final bound set.
\end{itemize}
\end{column}
\end{columns}
\medskip
{\small Also: configure before peak finding; restore particles exactly; order
corrected shapes as $0\leq c/a\leq b/a\leq1$.}
\end{frame}
