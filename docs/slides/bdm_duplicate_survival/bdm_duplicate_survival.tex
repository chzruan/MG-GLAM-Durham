\begin{frame}{Merge each strict duplicate group deterministically}
  \bodyfont
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
      \begin{panel}{bdmteal}
        {\color{bdmteal}\bfseries Require all four conditions}\par
        \[
          \begin{gathered}
            r_{ij}<0.2\,h^{-1}\mathrm{Mpc} \\[0.4ex]
            M_{\mathrm b,i}=M_{\mathrm b,j}\,,\qquad N_{\mathrm b,i}=N_{\mathrm b,j} \\[0.4ex]
            |\boldsymbol{v}_i-\boldsymbol{v}_j|<5\,\mathrm{km\,s^{-1}} \\[0.4ex]
            |\log_{10}M_{\mathrm{tot},i}-\log_{10}M_{\mathrm{tot},j}|<0.005
          \end{gathered}
        \]
        {\small Periodic separations; same strict criteria as Freyja.}
      \end{panel}
    \end{column}
    \begin{column}{0.47\textwidth}
      {\color{bdmblue}\bfseries One stable decision per group}\par
      \vspace{2mm}
      \begin{enumerate}\setlength\itemsep{1.5mm}
        \item Read unchanged measurements.
        \item Connect all strict pairs.
        \item Keep the \textbf{lowest original index} in each group.
        \item Then apply the removal mask.
      \end{enumerate}
    \end{column}
  \end{columns}
  \vspace{2mm}
  \begin{panel}{bdmblue}
    Same rule in the finder and cleaner; survivor properties are preserved.
  \end{panel}
  \evidence{Implementation: separate immutable host decision, followed by strict connected-component merging.}
  \note{The unequal-mass host test is a separate read-only pass with periodic
    geometry; its mask is applied before numerical-duplicate merging. The old
    parallel pass read and zeroed shared masses concurrently. Connected components
    handle transitive chains independently of linked-list traversal order.
    No unvalidated fractional-Rvir cut is added. Equal counts do not establish
    identical memberships. In-memory and rounded ASCII decisions can differ near
    thresholds. The upstream FI peak-generation race remains outside the patch;
    full historical host-set equivalence has not been established.}
\end{frame}
